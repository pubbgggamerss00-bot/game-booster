   466→   466→        get().freeRam(freed);
   467→   467→        set({ boost: { ...get().boost, netPing: net.ping, netJitter: net.jitter, quality, stage: 4, progress: 0 } });

   492→   492→        const ramAfter = get().perf.ramUsedGB;

   495→   495→          freedMB: freed,
   496→   496→          ramBefore: +ramBefore.toFixed(2),
   497→   497→          ramAfter: +ramAfter.toFixed(2),
   498→   498→          ping: net.ping,
   499→   499→          jitter: net.jitter,
   500→   500→          quality,
   501→   501→        };
   502→   502→        const entry: HistoryEntry = { id: uid(), kind: "boost", at: Date.now(), data: result };
